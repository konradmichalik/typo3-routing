# cat=basic/enable/010; type=string; label=Exclusive API path prefixes:Path spaces reserved exclusively for attribute routes (comma-separated, e.g. /api/, /va/, checked after stripping the site/language base). Inside them a path matching no route returns a JSON 404 instead of falling through to normal page rendering. This is NOT needed to make routes work: the performance gate that keeps matching off the hot path for ordinary page requests is derived from your #[Route] paths automatically. Leave EMPTY (default) unless you want that JSON 404 behaviour.
exclusivePrefixes =

# cat=authentication/enable/010; type=string; label=Bearer token env variable name:Default name of the process environment variable that holds the expected bearer token for BearerTokenAuthenticator. Only the variable NAME is stored here, never the secret value. Overridable per route via #[Authenticate(BearerTokenAuthenticator::class, options: ['envName' => '...'])]. Defaults to ROUTING_BEARER_TOKEN when left empty.
bearerTokenEnvName = ROUTING_BEARER_TOKEN

# cat=cors/enable/010; type=string; label=CORS allowed origins:Comma-separated list of origins allowed to call the API (e.g. https://app.example.com, https://admin.example.com). Use * to allow any origin. Leave EMPTY to disable CORS entirely (default). Applies to every matched attribute route; preflight OPTIONS requests are answered automatically.
cors.allowedOrigins =

# cat=cors/enable/020; type=string; label=CORS allowed request headers:Comma-separated list of request headers the browser may send (Access-Control-Allow-Headers). Defaults to Content-Type, Authorization.
cors.allowedHeaders = Content-Type, Authorization

# cat=cors/enable/030; type=boolean; label=CORS allow credentials:Allow credentialed requests (cookies, Authorization header). Requires an EXPLICIT origin list: combined with the * wildcard this setting is ignored (with a logged warning), because reflecting arbitrary origins with credentials would let any website read authenticated API responses.
cors.allowCredentials = 0

# cat=cors/enable/040; type=string; label=CORS exposed response headers:Comma-separated list of response headers exposed to the browser (Access-Control-Expose-Headers). Empty by default.
cors.exposeHeaders =

# cat=cors/enable/050; type=int+; label=CORS preflight max age:How long (seconds) the browser may cache the preflight result (Access-Control-Max-Age). Defaults to 3600.

Only ever reachable in the "Development" application context AND when this flag is enabled (both required) — never available in Production, regardless of this setting.
swaggerUi = 0
